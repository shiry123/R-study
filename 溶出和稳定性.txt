# 2. 计算传统f2因子 -------------------------------------------------------------
calculate_f2 <- function(ref, test, time_points) {
  # 确保没有缺失值
  valid <- !is.na(ref) & !is.na(test)
  ref <- ref[valid]
  test <- test[valid]
  
  # 计算差值平方和
  sum_sq_diff <- sum((ref - test)^2)
    # 计算f2
  f2 <- 50 * log10(1 / sqrt(1 + sum_sq_diff/length(time_points)) * 100)
  
  return(f2)
}

f2_classic <- calculate_f2(ref_mean_vec, test_mean_vec, time_points)

# 3. 拔靴法(Bootstrap)计算f2置信区间和相似概率 --------------------------------
bootstrap_f2 <- function(ref_data, test_data, n_bootstrap = 2000, conf_level = 0.9) {
  n_ref <- nrow(ref_data)
  n_test <- nrow(test_data)
  n_times <- ncol(ref_data)
  
  f2_boot <- numeric(n_bootstrap)
  
  for (i in 1:n_bootstrap) {
    # 有放回地重抽样
    ref_indices <- sample(1:n_ref, n_ref, replace = TRUE)
    test_indices <- sample(1:n_test, n_test, replace = TRUE)
    
    # 计算重抽样后的均值曲线
    ref_boot_mean <- colMeans(ref_data[ref_indices, ])
    test_boot_mean <- colMeans(test_data[test_indices, ])
    
    # 计算f2
    f2_boot[i] <- calculate_f2(ref_boot_mean, test_boot_mean, 1:n_times)
  }
  
  # 计算置信区间（百分位数法）
  alpha <- 1 - conf_level
  ci_lower <- quantile(f2_boot, alpha/2, na.rm = TRUE)
  ci_upper <- quantile(f2_boot, 1 - alpha/2, na.rm = TRUE)
  
  # 计算相似概率（f2 >= 50的概率）
  prob_similar <- mean(f2_boot >= 50, na.rm = TRUE)
  
  return(list(
    f2_bootstrap = f2_boot,
    ci_lower = ci_lower,
    ci_upper = ci_upper,
    prob_similar = prob_similar,
    mean_f2 = mean(f2_boot, na.rm = TRUE)
  ))
}

# 执行Bootstrap分析

    round(bootstrap_results$ci_upper, 2), "]\n", sep = "")
cat("相似概率P(f2 ≥ 50):", round(bootstrap_results$prob_similar * 100, 1), "%\n")

# 4. 马氏距离法(MSD) ----------------------------------------------------------
calculate_msd <- function(ref_data, test_data) {
  # 计算均值向量
  ref_mean <- colMeans(ref_data)
  test_mean <- colMeans(test_data)
  
  # 计算合并协方差矩阵
  n_ref <- nrow(ref_data)
  n_test <- nrow(test_data)
  cov_ref <- cov(ref_data)
  cov_test <- cov(test_data)
  pooled_cov <- ((n_ref - 1) * cov_ref + (n_test - 1) * cov_test) / (n_ref + n_test - 2)
  
  # 计算马氏距离
  diff_mean <- ref_mean - test_mean
  msd <- t(diff_mean) %*% solve(pooled_cov) %*% diff_mean
  
  # 转换为F统计量进行假设检验
  p <- ncol(ref_data)  # 变量数（时间点数）
  n <- n_ref + n_test
  f_stat <- (n_ref * n_test * (n - p - 1)) / (n * p * (n_ref + n_test - 2)) * msd
  df1 <- p
  df2 <- n - p - 1
  
  # 计算p值（双侧检验）
  p_value <- 1 - pf(f_stat, df1, df2)
  
  return(list(
    msd = as.numeric(msd),
    f_stat = f_stat,
    df1 = df1,
    df2 = df2,
    p_value = p_value
  ))
}
