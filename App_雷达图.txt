# 颜色生成函数
if (!requireNamespace("pals", quietly = TRUE)) {
  generate_colors <- function(n) {
    if (n <= 8) brewer.pal(n, "Set1") else rainbow(n)
  }
} else {
  generate_colors <- function(n) pals::cols25(n)[1:n]
}

# 阴影雷达图绘图函数
plot_shadow_radarchart <- function(rld_data, test_data = NULL, 
                                   RLD_label = "RLD均值±范围", 
                                   shadow_percent = 10,
                                   n_shadow = 20) {
  if (is.null(rld_data) || ncol(rld_data) == 0) {
    plot(1, type = "n", axes = FALSE, xlab = "", ylab = "", main = "无有效的参考组数据")
    text(1, 1, "请确保参考组包含至少一个完整样本", cex = 1.2, col = "red")
    return()
  }
  
  rld_mean <- apply(rld_data, 2, function(x) mean(x, na.rm = TRUE))
  n <- ncol(rld_data)
  
  shadow_range <- rbind(rld_mean * (1 - shadow_percent/100), 
                        rld_mean * (1 + shadow_percent/100))
  range_mat <- matrix(NA, nrow = n_shadow, ncol = n)
  for (i in 1:n) {
    range_mat[, i] <- seq(shadow_range[1, i], shadow_range[2, i], length.out = n_shadow)
  }
  range_df <- as.data.frame(range_mat)
  colnames(range_df) <- colnames(rld_data)
  
  all_data <- rbind(rld_data, test_data)
  global_max <- apply(all_data, 2, max, na.rm = TRUE) * 1.2
  global_min <- apply(all_data, 2, min, na.rm = TRUE) * 0.7
  
  radar_data <- rbind(global_max, global_min, range_df)
  if (!is.null(test_data) && nrow(test_data) > 0) {
    radar_data <- rbind(radar_data, test_data)
  }
  row_names <- c("max", "min", paste0("shadow", 1:n_shadow))
  if (!is.null(test_data) && nrow(test_data) > 0) {
    row_names <- c(row_names, rownames(test_data))
  }
  rownames(radar_data) <- row_names
  
  n_test <- ifelse(is.null(test_data), 0, nrow(test_data))
  if (n_test > 0) {
    test_colors <- generate_colors(n_test)
    test_plty <- rep(1, n_test)
    test_pty <- rep(16, n_test)
  } else {
    test_colors <- NULL
    test_plty <- NULL
    test_pty <- NULL
  }
  
  par(mar = c(2, 1, 1, 4), bg = "white")
  radarchart(radar_data,
             axistype = 2,
             seg = 5,
             pty = c(rep(32, n_shadow), test_pty),
             pcol = c(rep("#99CCFF", n_shadow), test_colors),
             plwd = 2,
             plty = c(rep(1, n_shadow), test_plty),
             cglcol = "grey", cglty = 1, axislabcol = "black", cglwd = 0.8,
             vlcex = 0.8)
  
  legend_labels <- c(RLD_label, if (n_test > 0) rownames(test_data) else NULL)
  legend_colors <- c("#99CCFF", if (n_test > 0) test_colors else NULL)
  legend("topright", legend = legend_labels, bty = "n", 
         pch = 20, col = legend_colors, text.col = "grey30", 
         cex = 0.9, pt.cex = 1.5)
}

# Server 逻辑
server <- function(input, output, session) {
  
  # 存储原始数据
  raw_data <- reactiveVal(NULL)
  # 标记数据是否已加载
  data_loaded <- reactiveVal(FALSE)
  
  # 当文件上传时，更新工作表选择器，并重置加载状态
  observeEvent(input$file, {
    req(input$file)
    sheet_names <- excel_sheets(input$file$datapath)
    output$sheet_ui <- renderUI({
      selectInput("sheet", "选择工作表", choices = sheet_names, selected = sheet_names[1])
    })
    raw_data(NULL)
    data_loaded(FALSE)
    # 清空动态生成的控件
    output$group_col_ui <- renderUI(NULL)
    output$label_col_ui <- renderUI(NULL)
    output$value_cols_ui <- renderUI(NULL)
    output$ref_group_ui <- renderUI(NULL)
    output$test_groups_ui <- renderUI(NULL)
  })
  
  # 点击“加载数据”按钮后读取工作表
  observeEvent(input$load_data, {
    req(input$file, input$sheet)
    tryCatch({
      dat <- read_excel(input$file$datapath, sheet = input$sheet)
      dat <- as.data.frame(dat)
      raw_data(dat)
      data_loaded(TRUE)
      
      # 动态生成列选择控件
      all_cols <- names(dat)
      
      group_default <- if ("Product" %in% all_cols) "Product" else all_cols[1]
      output$group_col_ui <- renderUI({
        selectInput("group_col", "分组列（例如 Product）", choices = all_cols, selected = group_default)
      })
      
      label_default <- if ("Batch" %in% all_cols) "Batch" else if ("Sample" %in% all_cols) "Sample" else all_cols[1]
      output$label_col_ui <- renderUI({
        selectInput("label_col", "标签列（图例显示，例如 Batch）", choices = all_cols, selected = label_default)
      })
      
      numeric_cols <- names(dat)[sapply(dat, is.numeric)]
      if (length(numeric_cols) == 0) {
        showNotification("未检测到数值列，无法绘制雷达图。", type = "error")
        output$value_cols_ui <- renderUI({
          selectInput("value_cols", "数值指标列（雷达图轴）", choices = NULL, multiple = TRUE)
        })
      } else {
        output$value_cols_ui <- renderUI({
          selectInput("value_cols", "数值指标列（雷达图轴）", choices = numeric_cols, multiple = TRUE,
                      selected = numeric_cols[1:min(5, length(numeric_cols))])
        })
      }
      
      showNotification("数据加载成功！", type = "message")
    }, error = function(e) {
      showNotification(paste("读取文件失败:", e$message), type = "error")
      raw_data(NULL)
      data_loaded(FALSE)
    })
  })
  
  # 当分组列改变时，更新参考组和测试组的下拉选项（仅在数据已加载时）
  observeEvent(c(input$group_col, raw_data()), {
    req(data_loaded(), raw_data(), input$group_col)
    group_vals <- unique(raw_data()[[input$group_col]])
    group_vals <- group_vals[!is.na(group_vals)]
    output$ref_group_ui <- renderUI({
      selectInput("ref_group", "参考组（均值阴影区域）", choices = group_vals, selected = group_vals[1])
    })
    output$test_groups_ui <- renderUI({
      selectInput("test_groups", "测试组（可多选）", choices = group_vals, selected = group_vals[2], multiple = TRUE)
    })
  })
  
  # 数据表预览（仅在数据加载后显示）
  output$data_table <- renderDT({
    req(data_loaded(), raw_data())
    datatable(raw_data(), options = list(scrollX = TRUE, pageLength = 10))
  })
  
  # 准备绘图数据（响应绘图按钮）
  plot_data <- eventReactive(input$plot_btn, {
    req(data_loaded(), raw_data(), input$group_col, input$label_col, input$value_cols, input$ref_group)
    dat <- raw_data()
    group_col <- input$group_col
    label_col <- input$label_col
    value_cols <- input$value_cols
    if (length(value_cols) < 3) {
      return(list(status = "error", message = "请至少选择3个数值指标列。"))
    }
    
    # 参考组数据
    ref_data <- dat[dat[[group_col]] == input$ref_group, value_cols, drop = FALSE]
    ref_data <- ref_data[complete.cases(ref_data), , drop = FALSE]
    if (nrow(ref_data) == 0) {
      return(list(status = "error", message = "参考组在选定的数值指标上无有效数据。"))
    }
    
    # 测试组数据
    test_groups <- input$test_groups
    if (is.null(test_groups) || length(test_groups) == 0) {
      test_data <- NULL
    } else {
      test_data_list <- list()
      for (grp in test_groups) {
        grp_data <- dat[dat[[group_col]] == grp, c(label_col, value_cols), drop = FALSE]
        grp_data <- grp_data[complete.cases(grp_data), , drop = FALSE]
        if (nrow(grp_data) > 0) {
          rownames(grp_data) <- grp_data[[label_col]]
          test_data_list[[grp]] <- grp_data[, value_cols, drop = FALSE]
        }
      }
      test_data <- if (length(test_data_list) > 0) do.call(rbind, test_data_list) else NULL
    }
    
    list(status = "success",
         rld_data = ref_data,
         test_data = test_data,
         ref_label = input$ref_group,
         value_cols = value_cols)
  })
  
  # 绘制雷达图
  output$radar_plot <- renderPlot({
    res <- plot_data()
    if (is.null(res) || res$status != "success") {
      plot(1, type = "n", axes = FALSE, xlab = "", ylab = "", main = "无法绘制雷达图")
      msg <- if (is.null(res)) "请点击「绘制雷达图」" else res$message
      text(1, 1, msg, cex = 1.2, col = "red")
      return()
    }
    
    plot_shadow_radarchart(
      rld_data = res$rld_data,
      test_data = res$test_data,
      RLD_label = paste0(res$ref_label, " 均值±", input$shadow_percent, "%"),
      shadow_percent = input$shadow_percent,
      n_shadow = input$n_shadow
    )
  }, height = 700, width = 800)
  
  # 状态文本
  output$plot_status <- renderText({
    if (!data_loaded()) return("请先选择文件和工作表，点击「加载数据」。")
    res <- plot_data()
    if (is.null(res)) return("请点击「绘制雷达图」。")
    if (res$status == "success") {
      n_test <- ifelse(is.null(res$test_data), 0, nrow(res$test_data))
      paste0("参考组: ", res$ref_label, " (", nrow(res$rld_data), " 个样本), ",
             "测试样本数: ", n_test, ", 雷达图指标: ", paste(res$value_cols, collapse = ", "))
    } else {
      res$message
    }
  })
}

shinyApp(ui, server)
