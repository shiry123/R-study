PBE=function(RLD,TEST,logData=FALSE){
  library(gridExtra)
  ###delete null columns
  #RLD <- RLD[,colSums(is.na(RLD))!= nrow(RLD),drop=F]
  #TEST <- TEST[,colSums(is.na(TEST))!= nrow(TEST),drop=F]
  ###delete null rows
  RLD<-na.omit(as.data.frame(RLD))
  TEST<-na.omit(as.data.frame(TEST))
  sigmaT0=0.1
  thetap=2.0891
  alpha=0.05
  m=length(colnames(RLD))
  
  ##ensure the data is logged, make RLD and TEST data ready for calculation
  if(isFALSE(logData)){
    RLDready = log(RLD)
    TESTready = log(TEST)
  } else {
    RLDready = RLD
    TESTready = TEST
  }
  
  ##calculate MSB and MSW
  MSB_MSW=function(x){
    if(m==1){
      MSB=sum((x-mean(c(as.matrix(x))))^2)/(nrow(x)-1) ##calculate MSB while m=1
      MSW=0
    }else{
      MSB=m*sum((apply(x,1,mean)-mean(c(as.matrix(x))))^2)/(nrow(x)-1)  
      ##calculate MSB while m>1, first calculate bottle mean, then total mean, then the sum of square of the difference
      MSW=sum((x-apply(x,1,mean))^2)/(nrow(x)*(m-1))
      ##calculate MSW, which only appears when m>2, first calculate bottle mean-each bottle, then sum of square of the difference 
    }
    return(cbind(MSB,MSW))
  }
  
  ##calculate sigmaR and sigmaT
  MSBR=MSB_MSW(RLDready)[1]
  MSBT=MSB_MSW(TESTready)[1]
  MSWR=MSB_MSW(RLDready)[2]
  MSWT=MSB_MSW(TESTready)[2]
  
  ##Point estimate  
  if(m==1){
    E2=0
    E4s=0
    E4c= 0
  }else{
    E2=(m-1)*MSWT/m
    E4s= -(1+thetap)*(m-1)*MSWR/m
    E4c=-(m-1)*MSWR/m
  }
  ############Reference-scale
  Ed=(mean(c(as.matrix(RLDready)))-mean(c(as.matrix(TESTready))))^2
  E1= MSBT/m
  E3s= -(1+thetap)*MSBR/m
  Eq1=sum(c(Ed,E1,E2,E3s,E4s))
  ############Constant-scale  
  E3c= -MSBR/m
  Eq2= sum(c(Ed,E1,E2,E3c,E4c,-thetap*sigmaT0^2))
  
  ##Confidence Bound and Uq
  if(m==1){
    H2= 0
    H4s= 0
    H4c= 0
    U2= 0
    U4s= 0
    U4c= 0
  }else{
    H2=  nrow(TESTready)*(m-1)*E2/qchisq(alpha, nrow(TESTready)*(m-1))
    H4s= nrow(RLDready)*(m-1)*E4s/qchisq(1-alpha, nrow(RLDready)*(m-1))
    H4c= nrow(RLDready)*(m-1)*E4c/qchisq(1-alpha, nrow(RLDready)*(m-1))
    U2= (H2-E2)^2
    U4s= (H4s-E4s)^2 
    U4c= (H4c-E4c)^2
  }
  ############Reference-scale
  Hd=(abs(mean(c(as.matrix(RLDready))) - mean(c(as.matrix(TESTready))))+
        qt(1-alpha, nrow(TESTready) + nrow(RLDready) - 2)*
        sqrt(MSBT/(nrow(TESTready)*m)+MSBR/(nrow(RLDready)*m)))^2
  H1= (nrow(TESTready)-1)*E1/qchisq(alpha, nrow(TESTready)-1)
  H3s= (nrow(RLDready)-1)*E3s/qchisq(1-alpha, nrow(RLDready)-1)
  Ud= (Hd-Ed)^2
  U1= (H1-E1)^2
  U3s= (H3s-E3s)^2
  ############Constant-scale  
  H3c= (nrow(RLDready)-1)*E3c/qchisq(1-alpha, nrow(RLDready)-1)
  U3c= (H3c-E3c)^2
  
  # Reference-scaled
  Uq1= sum(c(Ud,U1,U2,U3s,U4s))
  # Constant-scaled
  Uq2= sum(c(Ud,U1,U2,U3c,U4c))
  
  #H value
  Hn1= Eq1+sqrt(Uq1)
  Hn2= Eq2+sqrt(Uq2)
  
  ###conclusion
  sigmar=sqrt(MSBR/m+(m-1)*MSWR/m)
  sigmat=sqrt(MSBT/m+(m-1)*MSWT/m)
  SigmaR=c(sigmar,"Hn","Result")
  if (sigmar>0.1&Hn1<0){
    Reference=c("Y",Hn1,"Pass")
    Constant=c("N",Hn2,"")
  }else if(sigmar<0.1&Hn2<0){
    Reference=c("N",Hn1,"")
    Constant=c("Y",Hn2,"Pass")
  }else if(sigmar>0.1&Hn1>0){
    Reference=c("Y",Hn1,"Not Pass")
    Constant=c("N",Hn2,"")
  }else{
    Reference=c("N",Hn1,"")
    Constant=c("Y",Hn2,"Not Pass")
  }
  
  #summarise all data in one table
  R_Eq_intermediate_parameters=c(paste("Ed=",Ed),paste("E1=",E1),paste("E2=",E2),paste("E3s=",E3s),paste("E4s=",E4s),paste("Eq(linearized point estimate)=",Eq1))
  R_Hq_intermediate_parameters=c(paste("Hd=",Hd),paste("H1=",H1),paste("H2=",H2),paste("H3s=",H3s),paste("H4s=",H4s),"")
  R_Uq_intermediate_parameters=c(paste("Ud=",Ud),paste("U1=",U1),paste("U2=",U2),paste("U3s=",U3s),paste("U4s=",U4s),paste("Uq=(Hq-Eq)^2=",Uq1))
  R_Hn=c(rep("",5),paste("H1=",Hn1))
  
  C_Eq_intermediate_parameters=c(paste("Ed=",Ed),paste("E1=",E1),paste("E2=",E2),paste("E3c=",E3c),paste("E4c=",E4c),paste("Eq(linearized point estimate)=",Eq2))
  C_Hq_intermediate_parameters=c(paste("Hd=",Hd),paste("H1=",H1),paste("H2=",H2),paste("H3c=",H3c),paste("H4c=",H4c),"")
  C_Uq_intermediate_parameters=c(paste("Ud=",Ud),paste("U1=",U1),paste("U2=",U2),paste("U3c=",U3c),paste("U4c=",U4c),paste("Uq=(Hq-Eq)^2=",Uq2))
  C_Hn=c(rep("",5),paste("H2=",Hn2))
  
  Reference_scale=format(data.frame(R_Eq_intermediate_parameters,R_Hq_intermediate_parameters,R_Uq_intermediate_parameters,R_Hn),justify="left")
  Constant_scale=format(data.frame(C_Eq_intermediate_parameters,C_Hq_intermediate_parameters,C_Uq_intermediate_parameters,C_Hn),justify="left")
  Conclusion=format(data.frame(SigmaR,Reference,Constant))
  options(scipen=200)
  do.call(grid.arrange,c(list(tableGrob(Reference_scale),tableGrob(Constant_scale),tableGrob(Conclusion)),ncol=1))
  T;
}

require(gridExtra)
RLD<-dataset[dataset$Type=="RLD",]$ISM
TEST<-dataset[dataset$Type!="RLD",]$ISM

PBE(RLD,TEST)
