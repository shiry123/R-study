NO_GROUP <- "无"
NO_FILTER <- "无"

apply_trans <- function(x, method) {
  if (method == "none") return(x)
  if (method == "log10") {
    if (any(x <= 0, na.rm = TRUE)) warning("log10 变换要求数值 > 0，非正值已转为 NA")
    return(log10(x))
  }
  if (method == "ln") {
    if (any(x <= 0, na.rm = TRUE)) warning("自然对数变换要求数值 > 0，非正值已转为 NA")
    return(log(x))
  }
  x
}

server <- function(input, output, session) {
  
  data_raw <- reactiveVal()
  
  sheet_names <- reactive({
    req(input$file)
    excel_sheets(input$file$datapath)
  })
  
  output$sheet_ui <- renderUI({
    req(sheet_names())
    selectInput("sheet", "选择工作表", choices = sheet_names())
  })
  
  observeEvent(input$load_data, {
    req(input$file, input$sheet)
    tryCatch({
      df <- read_excel(input$file$datapath, sheet = input$sheet)
      data_raw(df)
      showNotification(paste("成功加载工作表:", input$sheet, " 行数:", nrow(df)), type = "message")
    }, error = function(e) {
      showNotification(paste("读取失败:", e$message), type = "error")
      data_raw(NULL)
    })
  })
  
  output$filter_var_ui <- renderUI({
    req(data_raw())
    choices <- c(NO_FILTER, names(data_raw()))
    selectInput("filter_var", "筛选变量（分类变量）", choices = choices, selected = NO_FILTER)
  })
  
  output$filter_value_ui <- renderUI({
    req(data_raw(), input$filter_var, input$filter_var != NO_FILTER)
    df <- data_raw()
    var <- input$filter_var
    unique_vals <- as.character(unique(df[[var]]))
    selectInput("filter_value", "选择值（可多选）", 
                choices = unique_vals, multiple = TRUE)
  })
  
  filtered_data <- reactive({
    req(data_raw(), input$filter_var)
    df <- data_raw()
    if (!is.null(input$filter_var) && input$filter_var != NO_FILTER && 
        !is.null(input$filter_value) && length(input$filter_value) > 0) {
      var <- input$filter_var
      df <- df[df[[var]] %in% input$filter_value, , drop = FALSE]
      if (nrow(df) == 0) {
        showNotification("筛选后无数据，请调整筛选条件", type = "warning")
      }
    }
    df
  })
  
  output$plot_type_ui <- renderUI({
    selectInput("plot_type", "图形类型",
                choices = c("散点图" = "scatter",
                            "分组折线图" = "line",
                            "簇状柱形图 (均值)" = "bar",
                            "箱型图" = "boxplot"))
  })
  
  output$xvar_ui <- renderUI({
    req(data_raw())
    selectInput("xvar", "X 轴变量", choices = names(data_raw()))
  })
  
  output$yvar_multi_ui <- renderUI({
    req(data_raw(), input$plot_type)
    num_cols <- names(data_raw())[sapply(data_raw(), is.numeric)]
    if (length(num_cols) == 0) num_cols <- names(data_raw())
    if (input$plot_type %in% c("scatter", "line")) {
      selectInput("yvar_multi", "Y 轴变量（可多选）", 
                  choices = num_cols, multiple = TRUE, selected = num_cols[1])
    } else {
      selectInput("yvar_single", "Y 轴变量 (需数值)", 
                  choices = num_cols, selected = num_cols[1])
    }
  })
  
  selected_yvars <- reactive({
    pt <- input$plot_type
    if (pt %in% c("scatter", "line")) {
      req(input$yvar_multi)
      input$yvar_multi
    } else {
      req(input$yvar_single)
      input$yvar_single
    }
  })
  
  output$groupvar_ui <- renderUI({
    req(data_raw())
    choices <- c(NO_GROUP, names(data_raw()))
    selectInput("groupvar", "分组变量 (颜色)", choices = choices, selected = NO_GROUP)
  })
  
  output$facet_ui <- renderUI({
    req(data_raw())
    selectInput("facet_vars", "分面变量 (可多选)", 
                choices = names(data_raw()), multiple = TRUE, selected = NULL)
  })
  
  output$trans_ui <- renderUI({
    req(input$plot_type)
    if (input$plot_type %in% c("scatter", "line")) {
      selectInput("trans", "X 轴变换",
                  choices = c("无" = "none", "log10" = "log10", "自然对数 (ln)" = "ln"),
                  selected = "none")
    } else {
      NULL
    }
  })
  
  output$extra_vars_ui <- renderUI({
    req(data_raw())
    selectInput("extra_vars", 
                "额外悬停信息变量（多选，仅散点/折线有效）", 
                choices = names(data_raw()), 
                multiple = TRUE)
  })
  
  output$data_table <- renderDT({
    req(data_raw())
    datatable(data_raw(), 
              options = list(scrollX = TRUE, pageLength = 10), 
              caption = "全部数据（可翻页）")
  })
  
  plot_obj <- reactive({
    req(filtered_data(), input$xvar, selected_yvars())
    df <- filtered_data()
    x_col <- input$xvar
    y_vars <- selected_yvars()
    group_col <- if (input$groupvar == NO_GROUP) NULL else input$groupvar
    facet_vars <- input$facet_vars
    plot_type <- input$plot_type
    x_angle <- input$x_angle
    
    apply_y_limits <- function(p) {
      y_min <- input$y_lim_min
      y_max <- input$y_lim_max
      if (!is.na(y_min) || !is.na(y_max)) {
        p <- p + coord_cartesian(ylim = c(y_min, y_max))
      }
      p
    }
    
    apply_angle <- function(p) {
      if (x_angle > 0) {
        p + theme(axis.text.x = element_text(angle = x_angle, hjust = 1, vjust = 0.5),
                  plot.margin = margin(b = 20, unit = "pt"))
      } else {
        p + theme(plot.margin = margin(b = 20, unit = "pt"))
      }
    }
    
    add_reference_lines <- function(p, x_is_numeric, trans_method, x_col, x_range = c(-Inf, Inf), y_range = c(-Inf, Inf)) {
      hline_str <- trimws(input$hline_values)
      vline_str <- trimws(input$vline_values)
      add_shadow <- input$add_shadow
      
      if (nchar(hline_str) > 0) {
        h_vals <- suppressWarnings(as.numeric(strsplit(hline_str, ",")[[1]]))
        h_vals <- h_vals[!is.na(h_vals)]
        if (length(h_vals) > 0) {
          if (add_shadow && length(h_vals) == 3) {
            p <- p + geom_rect(data = data.frame(xmin = x_range[1], xmax = x_range[2],
                                                 ymin = min(h_vals), ymax = max(h_vals)),
                               aes(xmin = xmin, xmax = xmax, ymin = ymin, ymax = ymax),
                               fill = "grey70", alpha = 0.4, inherit.aes = FALSE) +
              geom_hline(yintercept = h_vals, color = "green", linetype = "dashed", linewith = 0.5)
          } else {
            p <- p + geom_hline(yintercept = h_vals,
                                color = "blue", linetype = "dashed", linewith = 0.8)
          }
        }
      }
      
      if (x_is_numeric) {
        if (nchar(vline_str) > 0) {
          v_vals <- suppressWarnings(as.numeric(strsplit(vline_str, ",")[[1]]))
          v_vals <- v_vals[!is.na(v_vals)]
          if (length(v_vals) > 0) {
            if (trans_method != "none") {
              v_vals <- apply_trans(v_vals, trans_method)
              v_vals <- v_vals[is.finite(v_vals)]
            }
            if (length(v_vals) > 0) {
              if (add_shadow && length(v_vals) == 3) {
                p <- p + geom_rect(data = data.frame(xmin = min(v_vals), xmax = max(v_vals),
                                                     ymin = y_range[1], ymax = y_range[2]),
                                   aes(xmin = xmin, xmax = xmax, ymin = ymin, ymax = ymax),
                                   fill = "grey70", alpha = 0.4, inherit.aes = FALSE) +
                  geom_vline(xintercept = v_vals,
                             color = "green", linetype = "dashed", size = 0.5)
              } else {
                p <- p + geom_vline(xintercept = v_vals,
                                    color = "blue", linetype = "dashed", size = 0.8)
              }
            }
          }
        }
      } else {
        if (nchar(trimws(input$vline_values)) > 0) {
          showNotification("竖直线仅当 X 为数值变量时有效，当前 X 为因子/字符，已忽略", type = "warning")
        }
      }
      return(p)
    }
    
    if (!x_col %in% names(df)) return(NULL)
    if (!all(y_vars %in% names(df))) return(NULL)
    if (!is.null(group_col) && !group_col %in% names(df)) return(NULL)
    if (!is.null(facet_vars) && any(!facet_vars %in% names(df))) return(NULL)
    if (nrow(df) == 0) {
      showNotification("筛选后无数据", type = "error")
      return(NULL)
    }
    
    # ========== 柱形图 ==========
    if (plot_type == "bar") {
      y_var <- y_vars[1]
      if (length(y_vars) > 1) showNotification("柱形图仅支持单个 Y 变量，将使用第一个", type = "warning")
      plot_df <- data.frame(x = factor(df[[x_col]]), y = df[[y_var]])
      if (!is.null(group_col)) plot_df$group <- df[[group_col]]
      for (fv in facet_vars) plot_df[[fv]] <- df[[fv]]
      plot_df <- plot_df[!is.na(plot_df$y), ]
      if (nrow(plot_df) == 0) return(NULL)
      
      group_vars <- c("x")
      if (!is.null(group_col)) group_vars <- c(group_vars, "group")
      if (length(facet_vars) > 0) group_vars <- c(group_vars, facet_vars)
      
      summary_df <- plot_df %>%
        group_by(across(all_of(group_vars))) %>%
        summarise(mean = mean(y, na.rm = TRUE), sd = sd(y, na.rm = TRUE), n = n(), .groups = "drop") %>%
        mutate(se = sd / sqrt(n))
      
      p <- ggplot(summary_df, aes(x = x, y = mean))
      if (!is.null(group_col)) {
        p <- p + geom_col(aes(fill = group), position = position_dodge(0.9))
        if (input$bar_error) {
          error_type <- input$bar_error_type
          summary_df$error <- if (error_type == "sd") summary_df$sd else summary_df$se
          p <- p + geom_errorbar(aes(ymin = mean - error, ymax = mean + error, group = group),
                                 width = 0.2, position = position_dodge(0.9))
        }
        p <- p + labs(fill = group_col)
      } else {
        p <- p + geom_col(fill = "steelblue", position = position_dodge(0.9))
        if (input$bar_error) {
          error_type <- input$bar_error_type
          summary_df$error <- if (error_type == "sd") summary_df$sd else summary_df$se
          p <- p + geom_errorbar(aes(ymin = mean - error, ymax = mean + error),
                                 width = 0.2, position = position_dodge(0.9))
        }
      }
      p <- p + labs(x = x_col, y = y_var) + theme_bw()
      if (length(facet_vars) > 0) {
        facet_formula <- reformulate(paste(facet_vars, collapse = "+"))
        p <- p + facet_wrap(facet_formula, scales = "free_x")
      }
      p <- apply_y_limits(p)
      x_range <- c(-Inf, Inf)
      y_range <- range(summary_df$mean, na.rm = TRUE)
      p <- add_reference_lines(p, x_is_numeric = FALSE, trans_method = "none", x_col = x_col,
                               x_range = x_range, y_range = y_range)
      return(apply_angle(p))
    }
    
    # ========== 箱型图 ==========
    if (plot_type == "boxplot") {
      y_var <- y_vars[1]
      if (length(y_vars) > 1) showNotification("箱型图仅支持单个 Y 变量，将使用第一个", type = "warning")
      plot_df <- data.frame(x = factor(df[[x_col]]), y = df[[y_var]])
      if (!is.null(group_col)) plot_df$group <- df[[group_col]]
      for (fv in facet_vars) plot_df[[fv]] <- df[[fv]]
      plot_df <- plot_df[!is.na(plot_df$y), ]
      if (nrow(plot_df) == 0) return(NULL)
      
      p <- ggplot(plot_df, aes(x = x, y = y))
      if (!is.null(group_col)) {
        p <- p + geom_boxplot(aes(fill = group)) + labs(fill = group_col)
      } else {
        p <- p + geom_boxplot(fill = "steelblue")
      }
      p <- p + labs(x = x_col, y = y_var) + theme_bw()
      if (length(facet_vars) > 0) {
        facet_formula <- reformulate(paste(facet_vars, collapse = "+"))
        p <- p + facet_wrap(facet_formula, scales = "free_x")
      }
      p <- apply_y_limits(p)
      x_range <- c(-Inf, Inf)
      y_range <- range(plot_df$y, na.rm = TRUE)
      p <- add_reference_lines(p, x_is_numeric = FALSE, trans_method = "none", x_col = x_col,
                               x_range = x_range, y_range = y_range)
      return(apply_angle(p))
    }
    
    # ========== 散点图 & 折线图 ==========
    id_cols <- c(x_col)
    if (!is.null(group_col)) id_cols <- c(id_cols, group_col)
    if (length(facet_vars) > 0) id_cols <- c(id_cols, facet_vars)
    
    extra_vars <- input$extra_vars
    has_extra <- FALSE
    extra_var_names <- character(0)
    if (plot_type %in% c("scatter", "line") && !is.null(extra_vars) && length(extra_vars) > 0) {
      extra_vars <- intersect(extra_vars, names(df))
      if (length(extra_vars) > 0) {
        id_cols <- c(id_cols, extra_vars)
        has_extra <- TRUE
        extra_var_names <- extra_vars
      }
    }
    id_cols <- unique(id_cols)
    id_cols <- intersect(id_cols, names(df))
    
    df_long <- df[, c(id_cols, y_vars), drop = FALSE] %>%
      pivot_longer(cols = all_of(y_vars), names_to = "y_variable", values_to = "y_value") %>%
      filter(!is.na(y_value))
    
    if (nrow(df_long) == 0) {
      showNotification("长变换后无有效数据", type = "error")
      return(NULL)
    }
    
    trans_method <- input$trans
    x_is_numeric <- is.numeric(df_long[[x_col]])
    if (plot_type %in% c("scatter", "line") && trans_method != "none" && x_is_numeric) {
      zero_rows <- which(df_long[[x_col]] == 0)
      if (length(zero_rows) > 0) {
        df_long <- df_long[-zero_rows, ]
        showNotification(paste("删除 X=0 行数:", length(zero_rows)), type = "warning")
      }
      if (nrow(df_long) == 0) return(NULL)
      df_long$x_trans <- apply_trans(df_long[[x_col]], trans_method)
      df_long <- df_long[!is.na(df_long$x_trans), ]
      if (nrow(df_long) == 0) return(NULL)
      x_axis_val <- df_long$x_trans
      x_label <- paste0(switch(trans_method, "log10" = "log10", "ln" = "ln"), "(", x_col, ")")
    } else {
      x_axis_val <- df_long[[x_col]]
      x_label <- x_col
      if (trans_method != "none" && !x_is_numeric) 
        showNotification("X 变量非数值，无法应用变换", type = "warning")
    }
    df_long$x_axis <- x_axis_val
    
    # ---- 自定义悬停文本（避免重复显示 comb） ----
    df_long <- df_long %>%
      mutate(hover_text = paste0(
        "x: ", x_axis, "<br>",
        "y: ", y_value, "<br>",
        if (!is.null(group_col)) paste0("分组: ", .data[[group_col]], "<br>") else "",
        "Y变量: ", y_variable,
        if (has_extra) {
          extra_text <- df_long %>%
            select(all_of(extra_var_names)) %>%
            pmap_chr(function(...) {
              args <- list(...)
              paste(paste(extra_var_names, "=", args), collapse = "<br>")
            })
          paste0("<br>", extra_text)
        } else {
          ""
        }
      ))
    
    assign_comb_aesthetics <- function(df, group_col, y_vars) {
      comb_levels <- unique(interaction(df[[group_col]], df$y_variable, sep = "."))
      comb_names <- as.character(comb_levels)
      groups <- unique(df[[group_col]])
      group_colors <- hue_pal()(length(groups))
      names(group_colors) <- groups
      y_levels <- unique(df$y_variable)
      shapes <- 16:(16 + length(y_levels) - 1)
      linetypes <- 1:length(y_levels)
      names(shapes) <- y_levels
      names(linetypes) <- y_levels
      
      comb_colors <- sapply(comb_names, function(x) {
        grp <- strsplit(x, "\\.")[[1]][1]
        group_colors[grp]
      })
      comb_shapes <- sapply(comb_names, function(x) {
        yvar <- sub("^[^.]+\\.", "", x)
        shapes[yvar]
      })
      comb_linetypes <- sapply(comb_names, function(x) {
        yvar <- sub("^[^.]+\\.", "", x)
        linetypes[yvar]
      })
      names(comb_colors) <- comb_names
      names(comb_shapes) <- comb_names
      names(comb_linetypes) <- comb_names
      return(list(colors = comb_colors, shapes = comb_shapes, linetypes = comb_linetypes))
    }
    
    # ----- 散点图 -----
    if (plot_type == "scatter") {
      if (!x_is_numeric) {
        if (!is.null(group_col)) {
          pos <- position_dodge(width = 0.75)
        } else {
          pos <- position_jitter(width = 0.2, height = 0)
        }
      } else {
        pos <- "identity"
      }
      
      if (!is.null(group_col)) {
        df_long$comb <- interaction(df_long[[group_col]], df_long$y_variable, sep = ".")
        aes_map <- assign_comb_aesthetics(df_long, group_col, y_vars)
        
        p <- ggplot(df_long, aes(x = x_axis, y = y_value, 
                                 color = comb, shape = comb,
                                 group = comb,
                                 text = hover_text)) +
          geom_point(size = 2, position = pos) +
          scale_color_manual(values = aes_map$colors) +
          scale_shape_manual(values = aes_map$shapes) +
          labs(color = "分组:Y变量", shape = "分组:Y变量") +
          guides(color = guide_legend(order = 1), shape = guide_legend(order = 1))
      } else {
        p <- ggplot(df_long, aes(x = x_axis, y = y_value, 
                                 shape = y_variable,
                                 text = hover_text)) +
          geom_point(size = 2, color = "steelblue", position = pos) +
          labs(shape = "Y 变量") +
          guides(shape = guide_legend(order = 1))
      }
    }
    
    # ----- 折线图 -----
    if (plot_type == "line") {
      if (!is.null(group_col)) {
        df_long$comb <- interaction(df_long[[group_col]], df_long$y_variable, sep = ".")
        aes_map <- assign_comb_aesthetics(df_long, group_col, y_vars)
        
        p <- ggplot(df_long, aes(x = x_axis, y = y_value, 
                                 color = comb, linetype = comb, shape = comb,
                                 group = comb,
                                 text = hover_text)) +
          geom_line() + geom_point(size = 2) +
          scale_color_manual(values = aes_map$colors) +
          scale_linetype_manual(values = aes_map$linetypes) +
          scale_shape_manual(values = aes_map$shapes) +
          labs(color = "分组:Y变量", linetype = "分组:Y变量", shape = "分组:Y变量") +
          guides(color = guide_legend(order = 1), 
                 linetype = guide_legend(order = 1),
                 shape = guide_legend(order = 1))
      } else {
        df_long$group_id <- df_long$y_variable
        p <- ggplot(df_long, aes(x = x_axis, y = y_value, 
                                 linetype = y_variable, shape = y_variable,
                                 group = group_id,
                                 text = hover_text)) +
          geom_line(color = "steelblue") + 
          geom_point(size = 2, color = "steelblue") +
          labs(linetype = "Y 变量", shape = "Y 变量") +
          guides(linetype = guide_legend(order = 1), shape = guide_legend(order = 1))
      }
    }
    
    # 分面
    if (length(facet_vars) > 0) {
      facet_formula <- reformulate(paste(facet_vars, collapse = "+"))
      p <- p + facet_wrap(facet_formula, scales = "free_x")
    }
    
    p <- p + labs(x = x_label, y = "值") + theme_bw()
    p <- apply_y_limits(p)
    x_range <- range(df_long$x_axis, na.rm = TRUE)
    y_range <- range(df_long$y_value, na.rm = TRUE)
    p <- add_reference_lines(p, x_is_numeric = x_is_numeric, trans_method = trans_method, x_col = x_col,
                             x_range = x_range, y_range = y_range)
    return(apply_angle(p))
  })
  
  output$plot <- renderPlotly({
    p <- plot_obj()
    req(p)
    ggplotly(p, tooltip = "text") %>%   # 只显示自定义文本，屏蔽重复美学标签
      layout(margin = list(b = 150, l = 60, r = 30, t = 30))
  })
