plot_shadow_radarchart <- function(rld_data, test_data=NULL, RLD_label="RLD", n_shadow=20){
  library(fmsb)
  rld_mean <- round(apply(rld_data, 2, \(x) mean(x,na.rm = TRUE)),2)

  n=ncol(rld_data)
  range <-as.data.frame(matrix(1:n*n_shadow, nrow = n_shadow, ncol = n)) 
  colnames(range) <- colnames(rld_data)

  for (i in c(1:n)){
    range[i]<-c(seq(rld_mean[[i]]*0.9,rld_mean[[i]]*1.1,length=n_shadow))
  }
  
  radar_data <- as.data.frame(
    rbind(
      round(apply(rbind(rld_data,test_data), 2, \(x) max(x,na.rm = TRUE))*1.2,2),
      round(apply(rbind(rld_data,test_data), 2, \(x) min(x,na.rm = TRUE))*0.7,2),
      range,
      test_data
    )
  )
  
  row.names(radar_data) <- c("max","min",seq(1,n_shadow,1),rownames(test_data))
  
  if(is.null(test_data)){
    color_test=NULL
    plty_test=NULL
    pty_test=NULL
  }else{
    color_test=pals::cols25(nrow(test_data))
    plty_test=rep(1,nrow(test_data))    # 1线 0 点
    pty_test=rep(16,nrow(test_data))    # 20 小点 
  }
  
  par(mar = c(2, 1, 1, 4), bg = "white")
  radarchart(radar_data,
             axistype=2, 
             seg=5,
             pty=c(rep(32,n_shadow),pty_test),   
             pcol=c(rep("#99CCFF",n_shadow), color_test),   #  grey阴影
             plwd=2,
             plty=c(rep(1,n_shadow),plty_test), 
             cglcol="grey", cglty=1, axislabcol="black", cglwd=0.8,palcex=1
  )
  legend("topright",
         # x=1.3, y=1.2, 
         legend = c(RLD_label, rownames(test_data)),   # 阴影代表 RLD
         bty = "n", 
         pch=20 , 
         col=c("#99CCFF",color_test),
         text.col = "grey", 
         cex=1, 
         pt.cex=2
  )
  
}
