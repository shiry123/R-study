import xlwings as xw
import numpy as np
from scipy.optimize import curve_fit

# 定义四参数 Logistic 模型 (4PL)
def four_param_logistic(x, bottom, top, log_ec50, hill_slope):
    return bottom + (top - bottom) / (1 + 10**((log_ec50 - x) * hill_slope))

@xw.func
def calculate_ec50(concentrations, responses):
    """
    计算 EC50
    :param concentrations: Excel 中选中的浓度列 (一维数组)
    :param responses: Excel 中选中的响应值列 (一维数组)
    :return: EC50 数值
    """
    try:
        # 将数据转换为 numpy 数组并处理为浮点数
        x_data = np.array(concentrations, dtype=float)
        y_data = np.array(responses, dtype=float)

        # 过滤掉无效数据 (NaN 或 Inf)
        mask = np.isfinite(x_data) & np.isfinite(y_data)
        x_data = x_data[mask]
        y_data = y_data[mask]

        if len(x_data) < 4:
            return "错误: 数据点太少，至少需要4个点"

        # 对浓度取对数 (EC50 计算的标准步骤)
        # 注意：如果浓度有0或负数，这里会报错，需确保浓度为正数
        if np.any(x_data <= 0):
             return "错误: 浓度必须大于0"

        log_x = np.log10(x_data)

        # 初始参数猜测 [bottom, top, log_ec50, slope]
        # 简单的自动猜测逻辑
        p0 = [min(y_data), max(y_data), np.median(log_x), 1.0]

        # 执行拟合
        popt, pcov = curve_fit(four_param_logistic, log_x, y_data, p0=p0, maxfev=10000)

        # popt[2] 是 log_ec50，我们需要将其还原为真实浓度
        ec50_value = 10 ** popt[2]

        return round(ec50_value, 4) # 保留4位小数

    except Exception as e:
        # 捕获所有错误并返回给 Excel，防止 Excel 崩溃
        return f"计算出错: {str(e)}"
